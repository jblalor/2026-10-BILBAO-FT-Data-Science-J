---Enunciados de los ejercicios 
---1. 
SELECT DISTINCT E.Nombre
FROM reparto R
LEFT JOIN empleados E ON R.CodE = E.CodE
LEFT JOIN bares B ON R.CodB = B.CodB
WHERE R.Fecha BETWEEN '2005-10-17' AND '2005-10-23'
 AND B.Nombre = 'Stop';

---2. 
SELECT DISTINCT B.Cif, B.Nombre, B.Localidad
FROM reparto R
LEFT JOIN bares B ON R.CodB = B.CodB
LEFT JOIN cervezas C ON R.CodC = C.CodC
WHERE C.Envase = 'Botella' AND C.Capacidad < 1
ORDER BY B.Localidad;

---3. 
SELECT B.Nombre, C.Envase, C.Capacidad, R.Fecha, R.Cantidad
FROM reparto R
LEFT JOIN empleados E ON R.CodE = E.CodE
LEFT JOIN bares B ON R.CodB = B.CodB
LEFT JOIN cervezas C ON R.CodC = C.CodC
WHERE E.Nombre = 'Prudencio Caminero';

---4. 
SELECT DISTINCT B.Cif, B.Nombre, B.Localidad
FROM reparto R
LEFT JOIN bares B ON R.CodB = B.CodB
LEFT JOIN cervezas C ON R.CodC = C.CodC
WHERE C.Envase = 'Botella' AND C.Capacidad IN (0.2, 0.33);

---5. 
SELECT DISTINCT E.Nombre
FROM reparto R
LEFT JOIN empleados E ON R.CodE = E.CodE
LEFT JOIN bares B ON R.CodB = B.CodB
LEFT JOIN cervezas C ON R.CodC = C.CodC
WHERE B.Nombre IN ('Stop', 'Las Vegas') AND C.Envase = 'Botella';

---6. 
SELECT E.Nombre as 'Nombre empleado', 
  COUNT(*) AS 'Nº de viajes fuera de Villa Botijo'
FROM reparto R
LEFT JOIN empleados E ON R.CodE = E.CodE
LEFT JOIN bares B ON R.CodB = B.CodB
WHERE B.Localidad != 'Villa Botijo'
GROUP BY R.CodE;

---7. 
SELECT B.Nombre, B.Localidad, SUM(R.Cantidad * C.Capacidad) AS 'total litros cerveza comprados'
FROM reparto R
LEFT JOIN bares B ON R.CodB = B.CodB
LEFT JOIN cervezas C ON R.CodC = C.CodC
GROUP BY R.CodB
ORDER BY SUM(R.Cantidad * C.Capacidad) DESC
LIMIT 1;

---8. 
SELECT DISTINCT B.Nombre, B.Localidad
FROM reparto R
LEFT JOIN bares B ON R.CodB = B.CodB
LEFT JOIN cervezas C ON R.CodC = C.CodC
WHERE C.Envase = 'Botella' AND C.Capacidad < 1;

---9. 
UPDATE empleados
SET Sueldo = Sueldo * 1.05
WHERE CodE IN (SELECT CodE
FROM reparto R
GROUP BY CodE
ORDER BY COUNT(DISTINCT Fecha) DESC
LIMIT 1);
