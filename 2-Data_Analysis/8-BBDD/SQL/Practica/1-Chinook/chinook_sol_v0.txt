SEGUNDA PARTE 
1. 
SELECT C.FirstName, C.LastName, I.InvoiceId, I.InvoiceDate, I.BillingCountry
FROM customers C LEFT JOIN invoices I on C.CustomerId = I.InvoiceId
WHERE C.Country = 'Brazil';

2. 
SELECT I.InvoiceId, E.FirstName as 'nombre agente', E.LastName as 'apellido agente' FROM invoices I
LEFT JOIN customers C on I.CustomerId = C.CustomerId
LEFT JOIN employees E on C.SupportRepId = EmployeeId;

3. 
SELECT C.FirstName as 'nombre cliente', C.LastName as 'apellido cliente', C.Country as 'país cliente', E.FirstName as 'nombre agente', E.LastName as 'apellido agente', I.Total 
FROM invoices I
LEFT JOIN customers C on I.CustomerId = C.CustomerId
LEFT JOIN employees E on C.SupportRepId = EmployeeId;

4. 
SELECT I.InvoiceLineId as 'nº art.', I.InvoiceId as 'nº factura', T.Name as 'nombre canción' FROM invoice_items I
LEFT JOIN tracks T on I.TrackId = T.TrackId;

5. 
SELECT T.Name as 'track name', M.Name as 'format', A.Title as 'album', G.Name as 'genre'
from tracks T 
LEFT JOIN media_types M on T.MediaTypeId = M.MediaTypeId
LEFT JOIN albums A on T.AlbumId = A.AlbumId
LEFT JOIN genres G on T.GenreId = G.GenreId;

6. 
SELECT PlaylistId, Count(TrackId) FROM playlist_track

7. 
SELECT E.EmployeeId, E.LastName, E.FirstName, SUM(I.Total)
FROM invoices I 
LEFT JOIN customers C on I.CustomerId = C.CustomerId
LEFT JOIN employees E on C.SupportRepId = E.EmployeeId
GROUP BY E.EmployeeId;

8. El agente de ventas que más ha vendido en 2009 es Steve Johnson (total ventas = 164.34). 
SELECT E.EmployeeId, E.LastName, E.FirstName, SUM(I.Total)
FROM invoices I 
LEFT JOIN customers C on I.CustomerId = C.CustomerId
LEFT JOIN employees E on C.SupportRepId = E.EmployeeId
WHERE I.InvoiceDate LIKE '2009%'
GROUP BY E.EmployeeId;

9. 
SELECT AR.Name as 'grupo', SUM(I.UnitPrice * I.Quantity) as 'total'
FROM invoice_items I
LEFT JOIN tracks T on I.TrackId = T.TrackId
LEFT JOIN albums A on T.AlbumId = A.AlbumId
LEFT JOIN artists AR on A.ArtistId = AR.ArtistId
GROUP BY AR.ArtistId
ORDER BY total DESC
LIMIT 3;
